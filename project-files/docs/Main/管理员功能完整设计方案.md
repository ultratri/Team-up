## 1. 管理员角色定义

### 1.1 角色分类

基于综合方案，管理员应该分为三个层级：

```
平台管理员 (PLATFORM_ADMIN)
├── 最高权限
├── 管理所有数据
├── 系统配置
└── 用户管理

院系管理员 (DEPT_ADMIN)
├── 管理本院系数据
├── 审核本院系比赛
├── 查看本院系统计
└── 管理本院系用户

导师 (MENTOR)
├── 创建比赛
├── 管理自己的比赛
├── 指导学生团队
└── 评分权限
```

### 1.2 角色权限矩阵

| 功能模块 | 平台管理员 | 院系管理员 | 导师 | 普通用户 |
|---------|-----------|-----------|------|---------|
| 用户管理 | ✅ 全部 | ✅ 本院系 | ❌ | ❌ |
| 比赛管理 | ✅ 全部 | ✅ 本院系 | ✅ 自己的 | ❌ |
| 项目审核 | ✅ 全部 | ✅ 本院系 | ⭕ 关联的 | ❌ |
| 团队管理 | ✅ 全部 | ✅ 本院系 | ⭕ 指导的 | ✅ 自己的 |
| 数据统计 | ✅ 全部 | ✅ 本院系 | ⭕ 部分 | ❌ |
| 系统配置 | ✅ | ❌ | ❌ | ❌ |
| 技能标签库 | ✅ | ⭕ 建议 | ⭕ 建议 | ❌ |
| 公告发布 | ✅ 全部 | ✅ 本院系 | ❌ | ❌ |
| 举报处理 | ✅ 全部 | ✅ 本院系 | ❌ | ❌ |

### 3.1 团队管理模块

#### 3.1.1 为什么需要？

基于混合模型，团队分为临时和长期两种：
- **临时团队**：项目结束自动解散，无需管理
- **长期团队**：需要审核和管理，防止滥用

**场景：**
```
某些学生创建"长期团队"，但实际上是空壳团队
→ 占用系统资源
→ 影响匹配质量
→ 需要管理员审核和清理
```

**后端接口**
```java
/**
 * 团队管理接口
 */
@RestController
@RequestMapping("/api/admin/teams")
@PreAuthorize("hasAnyRole('PLATFORM_ADMIN', 'DEPT_ADMIN')")
public class AdminTeamController {
    
    /**
     * 获取团队列表
     */
    @GetMapping
    public Result<Page<Team>> listTeams(
        @RequestParam(defaultValue = "1") int page,
        @RequestParam(defaultValue = "20") int size,
        @RequestParam(required = false) String type,
        @RequestParam(required = false) Boolean isActive,
        @RequestParam(required = false) String keyword
    ) {
        // 院系管理员只能看本院系的团队
        Long adminId = UserContext.getCurrentUserId();
        String adminRole = UserContext.getCurrentUserRole();
        
        LambdaQueryWrapper<Team> wrapper = new LambdaQueryWrapper<>();
        
        if ("DEPT_ADMIN".equals(adminRole)) {
            // 获取管理员的院系
            String department = userService.getUserDepartment(adminId);
            // 只查询本院系的团队
            wrapper.in(Team::getLeaderId, 
                userService.getUserIdsByDepartment(department));
        }
        
        if (type != null) {
            wrapper.eq(Team::getType, type);
        }
        if (isActive != null) {
            wrapper.eq(Team::getIsActive, isActive);
        }
        if (keyword != null) {
            wrapper.like(Team::getName, keyword);
        }
        
        Page<Team> result = teamService.page(new Page<>(page, size), wrapper);
        return Result.success(result);
    }
    
    /**
     * 解散团队
     */
    @PostMapping("/{id}/dissolve")
    public Result<Void> dissolveTeam(
        @PathVariable Long id,
        @RequestBody Map<String, String> params
    ) {
        String reason = params.get("reason");
        
        Team team = teamService.getById(id);
        if (team == null) {
            return Result.error(404, "团队不存在");
        }
        
        // 检查权限
        if (!hasPermission(team)) {
            return Result.error(403, "无权操作");
        }
        
        // 解散团队
        team.setIsActive(false);
        teamService.updateById(team);
        
        // 记录审计日志
        auditLogService.log("DISSOLVE_TEAM", "TEAM", id, 
            "管理员解散团队：" + team.getName() + "，原因：" + reason);
        
        // 通知团队成员
        notificationService.notifyTeamMembers(id, 
            "团队已被管理员解散", reason);
        
        return Result.success();
    }
}
```

```java
     */

        return Result.success();
```

```java
            "导师申请已通过", 
            "恭喜！您已成为平台导师");
        
        return Result.success();
    }
```

```java
    }
    
    /**
     * 审核项目
     */
    @PostMapping("/projects/{id}/approve")
    public Result<Void> approveProject(@PathVariable Long id) {
        Project project = projectService.getById(id);
        project.setStatus("APPROVED");
        project.setReviewedBy(UserContext.getCurrentUserId());
        project.setReviewedAt(LocalDateTime.now());
        projectService.updateById(project);
        
        // 通知项目创建者
        notificationService.notify(project.getCreatorId(), 
            "项目审核通过", 
            "您的项目"" + project.getName() + ""已通过审核");

    }
    
    /**
     * 获取举报列表
     */
    @GetMapping("/reports")
    public Result<Page<Report>> getReports(
        @RequestParam(defaultValue = "1") int page,
        @RequestParam(defaultValue = "20") int size,
        @RequestParam(required = false) String status
    ) {
        Page<Report> result = reportService.getReports(page, size, status);
        return Result.success(result);
    }
    
    /**
     * 处理举报
     */
    @PostMapping("/reports/{id}/handle")
    public Result<Void> handleReport(
        @PathVariable Long id,
        @RequestBody Map<String, String> params
    ) {
        String action = params.get("action"); // VALID, INVALID
        String reason = params.get("reason");
        
        Report report = reportService.getById(id);
        report.setStatus(action);
        report.setHandledBy(UserContext.getCurrentUserId());
        report.setHandledAt(LocalDateTime.now());
        report.setHandleReason(reason);
        reportService.updateById(report);
        
        if ("VALID".equals(action)) {
            // 举报有效，对被举报对象进行处理
            handleValidReport(report);
        }
        
        // 通知举报人
        notificationService.notify(report.getReporterId(), 
            "举报处理结果", 
            "您的举报已处理：" + reason);
        
        return Result.success();
    }
    
    private void handleValidReport(Report report) {
        if ("USER".equals(report.getTargetType())) {
            // 处理用户：警告、禁言、封号
            User user = userService.getById(report.getTargetId());
            user.setWarningCount(user.getWarningCount() + 1);
            if (user.getWarningCount() >= 3) {
                user.setStatus("BANNED");
            }
            userService.updateById(user);
        } else if ("PROJECT".equals(report.getTargetType())) {
            // 处理项目：下架、删除
            Project project = projectService.getById(report.getTargetId());
            project.setStatus("REMOVED");
            projectService.updateById(project);
        }
    }
}
```

```java
    /**
     * 下载导出文件
     */
    @GetMapping("/{id}/download")
    public void downloadExport(
        @PathVariable Long id,
        HttpServletResponse response
    ) throws IOException {
        ExportTask task = exportTaskService.getById(id);
        
        // 检查权限
        if (!task.getCreatedBy().equals(UserContext.getCurrentUserId())) {
            response.setStatus(403);
            return;
        }
        
        // 下载文件
        File file = new File(task.getFilePath());
        response.setContentType("application/octet-stream");
        response.setHeader("Content-Disposition", 
            "attachment; filename=" + task.getFileName());
        
        try (InputStream in = new FileInputStream(file);
             OutputStream out = response.getOutputStream()) {
            byte[] buffer = new byte[4096];
            int length;
            while ((length = in.read(buffer)) > 0) {
                out.write(buffer, 0, length);
            }
        }
    }
}
```

```java
/**
 * 用户管理接口（管理员）
 */
@RestController
@RequestMapping("/api/admin/users")
@PreAuthorize("hasRole('PLATFORM_ADMIN')")
public class AdminUserController {
    
    @Autowired
    private UserService userService;
    
    @Autowired
    private UserProfileService userProfileService;
    
    @Autowired
    private UserReputationService userReputationService;
    
    @Autowired
    private AuditLogService auditLogService;
    
    /**
     * 获取用户列表
     */
    @GetMapping
    public Result<Page<UserVO>> listUsers(
        @RequestParam(defaultValue = "1") Integer page,
        @RequestParam(defaultValue = "10") Integer size,
        @RequestParam(required = false) String role,
        @RequestParam(required = false) String status,
        @RequestParam(required = false) String department,
        @RequestParam(required = false) String keyword
    ) {
        LambdaQueryWrapper<User> wrapper = new LambdaQueryWrapper<>();
        
        if (StringUtils.isNotBlank(role)) {
            wrapper.eq(User::getRole, role);
        }
        if (StringUtils.isNotBlank(status)) {
            wrapper.eq(User::getStatus, status);
        }
        if (StringUtils.isNotBlank(keyword)) {
            wrapper.and(w -> w
                .like(User::getStudentId, keyword)
                .or().like(User::getUsername, keyword)
                .or().like(User::getEmail, keyword)
            );
        }
        
        Page<User> userPage = userService.page(
            new Page<>(page, size),
            wrapper.orderByDesc(User::getCreatedAt)
        );
        
        // 转换为VO，包含用户档案信息
        Page<UserVO> voPage = userPage.convert(user -> {
            UserVO vo = new UserVO();
            BeanUtils.copyProperties(user, vo);
            
            // 获取用户档案
            UserProfile profile = userProfileService.getByUserId(user.getId());
            if (profile != null) {
                vo.setRealName(profile.getRealName());
                vo.setDepartment(profile.getDepartment());
                vo.setMajor(profile.getMajor());
                vo.setGrade(profile.getGrade());
            }
            
            // 获取信誉信息
            UserReputation reputation = userReputationService.getByUserId(user.getId());
            if (reputation != null) {
                vo.setReputationScore(reputation.getReputationScore());
                vo.setCompletedProjectsCount(reputation.getCompletedProjectsCount());
            }
            
            return vo;
        });
        
        return Result.success(voPage);
    }
    
    /**
     * 修改用户角色
     */
    @PutMapping("/{userId}/role")
    public Result<Void> updateUserRole(
        @PathVariable Long userId,
        @RequestBody UpdateRoleRequest request
    ) {
        User user = userService.getById(userId);
        if (user == null) {
            return Result.error(404, "用户不存在");
        }
        
        String oldRole = user.getRole();
        String newRole = request.getNewRole();
        
        user.setRole(newRole);
        userService.updateById(user);
        
        // 记录审计日志
        auditLogService.logSensitiveOperation(
            "UPDATE_USER_ROLE",
            "USER",
            userId,
            String.format("修改用户角色：%s -> %s", oldRole, newRole),
            "SUCCESS",
            null
        );
        
        return Result.success();
    }
    
    /**
     * 禁用/启用用户
     */
    @PutMapping("/{userId}/status")
    public Result<Void> toggleUserStatus(@PathVariable Long userId) {
        User user = userService.getById(userId);
        if (user == null) {
            return Result.error(404, "用户不存在");
        }
        
        String newStatus = "ACTIVE".equals(user.getStatus()) ? "BANNED" : "ACTIVE";
        user.setStatus(newStatus);
        userService.updateById(user);
        
        // 记录审计日志
        auditLogService.logSensitiveOperation(
            "TOGGLE_USER_STATUS",
            "USER",
            userId,
            "ACTIVE".equals(newStatus) ? "启用用户" : "禁用用户",
            "SUCCESS",
            null
        );
        
        return Result.success();
    }
    
    /**
     * 重置用户密码
     */
    @PostMapping("/{userId}/reset-password")
    public Result<String> resetPassword(@PathVariable Long userId) {
        User user = userService.getById(userId);
        if (user == null) {
            return Result.error(404, "用户不存在");
        }
        
        // 生成随机密码
        String newPassword = generateRandomPassword();
        user.setPassword(passwordEncoder.encode(newPassword));
        userService.updateById(user);
        
        // 记录审计日志
        auditLogService.logSensitiveOperation(
            "RESET_PASSWORD",
            "USER",
            userId,
            "重置用户密码",
            "SUCCESS",
            null
        );
        
        // 发送通知
        notificationService.notify(userId, 
            "密码已重置", 
            "您的密码已被管理员重置，新密码为：" + newPassword);
        
        return Result.success(newPassword);
    }
    
    private String generateRandomPassword() {
        String chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
        StringBuilder password = new StringBuilder();
        Random random = new Random();
        for (int i = 0; i < 10; i++) {
            password.append(chars.charAt(random.nextInt(chars.length())));
        }
        return password.toString();
    }
}
```

```typescript
// router/index.ts
router.beforeEach((to, from, next) => {
  const userStore = useUserStore();
  const role = userStore.role;
  
  // 管理员路由
  if (to.path.startsWith('/admin')) {
    if (role !== 'PLATFORM_ADMIN') {
      ElMessage.error('无权访问');
      next('/');
      return;
    }
  }
  
  // 导师路由
  if (to.path.startsWith('/mentor')) {
    if (role !== 'MENTOR' && role !== 'PLATFORM_ADMIN') {
      ElMessage.error('无权访问');
      next('/');
      return;
    }
  }
  
  next();
});
```
